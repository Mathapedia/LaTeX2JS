\section{The Cauchy--Schwarz Inequality}

\begin{theorem}
If $u, v \in V$ then $|\langle u, v \rangle| \le \|u\| \|v\|$.
\end{theorem}

\begin{proof}
Consider the quadratic polynomial
$$p(t) = \langle u + t v, u + t v \rangle \ge 0.$$
\end{proof}
